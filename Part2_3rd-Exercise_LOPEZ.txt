        {

        }
        private void btnFight_Click(object sender, EventArgs e)
        {
            int door;

            // TryParse prevents application crashes if the user types letters instead of numbers
            if (int.TryParse(txtDoor.Text, out door))
            {
                switch (door)
                {
                    case 1:
                        lblMessage.Text = "💰 You found treasure!";
                        break;

                    case 2:
                        lblMessage.Text = "👾 A monster appeared!";
                        break;

                    case 3:
                        lblMessage.Text = "🧪 You found a magic potion!";
                        break;

                    default:
                        lblMessage.Text = "Choose Door 1, 2, or 3.";
                        break;
                }
            }
            else
            {
                lblMessage.Text = "Please enter a valid number (1, 2, or 3)!";
            }
        }
